function Scan({ go }: { go: (screen: Screen) => void }) {
  const videoRef = useRef<HTMLVideoElement>(null);
  const [cameraState, setCameraState] = useState<
    "idle" | "active" | "denied" | "invalid"
  >("idle");

  useEffect(() => {
    return () => {
      const stream = videoRef.current?.srcObject as MediaStream | null;
      stream?.getTracks().forEach((track) => track.stop());
    };
  }, []);

  const startCamera = async () => {
    try {
      const stream = await navigator.mediaDevices.getUserMedia({
        video: { facingMode: "environment" },
      });
      if (videoRef.current) {
        videoRef.current.srcObject = stream;
        await videoRef.current.play();
      }
      setCameraState("active");
    } catch {
      setCameraState("denied");
    }
  };

  return (
    <main className="app-shell scan-screen">
      <header className="scan-header">
        <button
          className="back-button"
          onClick={() => go("home")}
          type="button"
          aria-label="ホームに戻る"
        >
          <Back />
        </button>
        <Logo compact />
        <span className="header-spacer" />
      </header>
      <div className="content-scroll">
        <div className="screen scan-content">
          <div className="title-block center">
            <span className="kicker">QR SCAN</span>
            <h1>ルームに参加</h1>
            <p>参加するルームのQRコードを読み取ってください</p>
          </div>

          <section className="camera-card">
            <video
              ref={videoRef}
              className={`camera-video ${cameraState === "active" ? "visible" : ""}`}
              muted
              playsInline
              aria-label="QRコード読み取り用カメラプレビュー"
            />
            <div className="camera-overlay">
              <div className="scan-frame">
                <i /><i /><i /><i />
                {cameraState === "active" && <span className="scan-line" />}
              </div>
            </div>
            {cameraState !== "active" && (
              <div className="camera-placeholder">
                <span><Camera /></span>
                {cameraState === "idle" ? (
                  <>
                    <strong>カメラを起動してください</strong>
                    <small>QRコードの読み取りにのみ使用します</small>
                    <button onClick={startCamera} type="button">カメラを許可する</button>
                  </>
                ) : (
                  <>
                    <strong>
                      {cameraState === "denied"
                        ? "カメラを利用できません"
                        : "有効なQRコードではありません"}
                    </strong>
                    <small>
                      {cameraState === "denied"
                        ? "端末の設定からカメラへのアクセスを許可してください"
                        : "メシマッチの招待QRコードを読み取ってください"}
                    </small>
                    <button onClick={startCamera} type="button">もう一度試す</button>
                  </>
                )}
              </div>
            )}
          </section>

          {cameraState === "active" && (
            <div className="scan-status">
              <span className="scan-dot" />
              QRコードを探しています…
              <button onClick={() => setCameraState("invalid")} type="button">
                読み取り失敗
              </button>
            </div>
          )}

          <div className="demo-join">
            <div>
              <strong>カメラが使えない場合</strong>
              <span>デモ用の招待から次の画面を確認できます</span>
            </div>
            <Button variant="secondary" onClick={() => go("join")}>
              <QrIcon /> デモ用ルームに参加する <Arrow />
            </Button>
          </div>

          <button className="text-back" onClick={() => go("home")} type="button">
            <Back /> ホームに戻る
          </button>
        </div>
      </div>
    </main>
  );
}

export default function App() {
  const [screen, setScreen] = useState<Screen>(() => {
    const isInviteUrl =
      window.location.pathname.startsWith("/j/") ||
      new URLSearchParams(window.location.search).has("room");
    return isInviteUrl ? "join" : "home";
  });
  const [hostName, setHostName] = useState("");
  const [participantName, setParticipantName] = useState("");
  const home = () => setScreen("home");
  if (screen === "home") return <Home go={setScreen} />;
  if (screen === "scan") return <Scan go={setScreen} />;
  if (screen === "create")
    return (
      <Create
        go={setScreen}
        home={home}
        hostName={hostName}
        setHostName={setHostName}
      />
    );
  if (screen === "waiting")
    return <Waiting go={setScreen} home={home} hostName={hostName} />;
  if (screen === "join")
    return (
      <Join
        go={setScreen}
        home={home}
        name={participantName}
        setName={setParticipantName}
      />
    );
  if (screen === "participant-waiting")
    return (
      <Waiting
        go={setScreen}
        home={home}
        hostName="ゆうき"
        isHost={false}
        participantName={participantName}
      />
    );
  if (screen === "genre") return <GenreVote go={setScreen} home={home} />;
  if (screen === "vote-wait")
    return <VoteWait go={setScreen} home={home} next="genre-result" />;
  if (screen === "shop-wait")
    return <VoteWait go={setScreen} home={home} next="final" />;
  if (screen === "genre-result") return <GenreResult go={setScreen} home={home} />;
  if (screen === "shop") return <ShopVote go={setScreen} home={home} />;
  return <Final home={home} />;
}
